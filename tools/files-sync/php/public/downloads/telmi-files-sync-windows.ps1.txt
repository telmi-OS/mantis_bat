# Telmi Files Sync for Windows — Ghost API v2, Windows PowerShell 5.1 (built in).
# No Python, curl, WSL, modules, installation or administrator access required.
# Usage:
#   powershell.exe -NoProfile -File .\telmi-files-sync-windows.ps1 [--dry-run] [--pull] [--push] [--verbose]
#
# Contract: SERVER WINS. Identity-aware moves; archive-first local deletes; direct remote-to-local deletes. Remote inventory MUST contain
# at least one file. HTTP/API/validation/transfer failures HARD STOP.
# Existing files: file_replace (file_id + expected_sha256); new: upload.
# GETs may retry; mutation POSTs are NEVER retried automatically.
#
# Keep credentials secret. If you configure a JWT here, restrict this file's ACL.

Set-StrictMode -Version Latest
$ErrorActionPreference = 'Stop'
$ProgressPreference = 'SilentlyContinue'

# --- CONFIGURATION -----------------------------------------------------------
# Environment variables with the same names override these defaults.
$Config = @{
    TELMI_BASE_URL     = 'https://YOUR-TELMI-HOST.example'
    TELMI_SPACE_ID     = 'PASTE_YOUR_SPACE_ID_HERE'
    TELMI_GHOST_JWT    = 'PASTE_YOUR_GHOST_JWT_HERE'
    SYNC_DIR           = (Join-Path $env:USERPROFILE 'TelmiSync\MySpace')
    STATE_FILE         = '.telmi-sync-state.json'
    LOG_DIR            = (Join-Path $env:LOCALAPPDATA 'TelmiFilesSync\Logs')
    LOCK_DIR           = (Join-Path $env:LOCALAPPDATA 'TelmiFilesSync\Locks')
    HTTP_RETRIES       = 3                # GET requests only
    HTTP_RETRY_SLEEP   = 3                # seconds
    HTTP_READ_TIMEOUT  = 120              # seconds
    HTTP_TRANSFER_TIMEOUT = 3600          # seconds
    LIST_PAGE_SIZE     = 500              # API maximum; no pagination available
    MAX_FOLDER_DEPTH   = 32
    MAX_LOCAL_DELETES_PER_RUN = 10
    MAX_LOCAL_DELETE_PERCENT = 25
}
foreach ($key in @($Config.Keys)) {
    $override = [Environment]::GetEnvironmentVariable($key, 'Process')
    if (-not [string]::IsNullOrEmpty($override)) { $Config[$key] = $override }
}

# --- HELPERS -----------------------------------------------------------------
function Log([string]$Message) {
    $line = '{0} {1}' -f ([DateTimeOffset]::Now.ToString('yyyy-MM-ddTHH:mm:sszzz')), $Message
    [Console]::Error.WriteLine($line)
    if ($script:LogPath) {
        [IO.File]::AppendAllText($script:LogPath, "$line`r`n", (New-Object System.Text.UTF8Encoding($false)))
    }
}
function Stop-Sync([string]$Message) { throw [InvalidOperationException]::new($Message) }
function Prop($Obj, [string]$Name) {
    if ($null -eq $Obj) { return $null }
    if ($Obj -is [System.Collections.IDictionary]) { return ,$Obj[$Name] }
    $p = $Obj.PSObject.Properties[$Name]
    if ($null -eq $p) { return $null }
    return ,$p.Value
}
function Sha([string]$Path) {
    $stream = [IO.File]::Open($Path, [IO.FileMode]::Open, [IO.FileAccess]::Read, [IO.FileShare]::Read)
    $algo = [Security.Cryptography.SHA256]::Create()
    try { return ([BitConverter]::ToString($algo.ComputeHash($stream))).Replace('-', '').ToLowerInvariant() }
    finally { $algo.Dispose(); $stream.Dispose() }
}
function Is-Sha($Value) { return ($Value -is [string] -and $Value -cmatch '^[0-9a-fA-F]{64}$') }
function Canon([string]$Name) { return $Name.Normalize([Text.NormalizationForm]::FormC).ToUpperInvariant() }
function Check-Name($Name, [string]$Label) {
    if ($Name -isnot [string] -or $Name.Length -eq 0 -or $Name -eq '.' -or $Name -eq '..' -or
        $Name -match '[<>:"/\\|?*\x00-\x1f]' -or $Name.EndsWith(' ') -or $Name.EndsWith('.') -or
        $Name -match '^(?i:(CON|PRN|AUX|NUL|CONIN\$|CONOUT\$|COM[1-9]|LPT[1-9]))(\.|$)') {
        Stop-Sync "unsafe Windows $Label name: '$Name'"
    }
    return $Name
}
function Parse-Int([string]$Key, [int]$Min, [int]$Max) {
    $n = 0
    if (-not [int]::TryParse([string]$Config[$Key], [ref]$n) -or $n -lt $Min -or $n -gt $Max) {
        Stop-Sync "invalid $Key (expected integer $Min..$Max)"
    }
    return $n
}
function Rel-Path([string]$Rel) {
    $path = $script:Root
    foreach ($part in $Rel.Split('/')) {
        $null = Check-Name $part 'path component'
        $path = [IO.Path]::Combine($path, $part)
        if ([IO.File]::Exists($path) -or [IO.Directory]::Exists($path)) { Assert-NotLink $path }
    }
    $full = [IO.Path]::GetFullPath($path)
    $prefix = $script:Root.TrimEnd('\') + '\'
    if (-not $full.StartsWith($prefix, [StringComparison]::OrdinalIgnoreCase)) {
        Stop-Sync "path escapes sync root: '$Rel'"
    }
    return $full
}
function Assert-NotLink([string]$Path) {
    $attr = [IO.File]::GetAttributes($Path)
    if (($attr -band [IO.FileAttributes]::ReparsePoint) -ne 0) {
        Stop-Sync "reparse point/junction/symlink refused: $Path"
    }
}
function Parent-Rel([string]$Rel) { $i = $Rel.LastIndexOf('/'); if ($i -lt 0) { return '' }; return $Rel.Substring(0, $i) }
function Leaf-Rel([string]$Rel) { $i = $Rel.LastIndexOf('/'); return $Rel.Substring($i + 1) }
function New-Dict { return [Collections.Generic.Dictionary[string,object]]::new([StringComparer]::Ordinal) }
function File-CopyChecked([string]$From, [string]$To, [string]$Expected) {
    [IO.File]::Copy($From, $To, $false)
    if ((Sha $To) -cne $Expected) { Stop-Sync "file changed during snapshot/copy: $From" }
}
function Replace-Atomic([string]$Temp, [string]$Dest) {
    if ([IO.Directory]::Exists($Dest)) { Stop-Sync "destination became a directory: $Dest" }
    if ([IO.File]::Exists($Dest)) {
        Assert-NotLink $Dest
        [IO.File]::Replace($Temp, $Dest, [NullString]::Value)
    } else {
        [IO.File]::Move($Temp, $Dest)
    }
}
function Save-State {
    if ($script:DryRun) { return }
    [IO.Directory]::CreateDirectory($script:Root) | Out-Null
    $dest = Rel-Path ([string]$Config.STATE_FILE)
    $tmp = Join-Path $script:Root ('.telmi-state-' + [Guid]::NewGuid().ToString('N'))
    try {
        $json = $script:Manifest | ConvertTo-Json -Depth 30
        $bytes = [Text.UTF8Encoding]::new($false).GetBytes($json + "`n")
        $stream = [IO.File]::Open($tmp, 'CreateNew', 'Write', 'None')
        try { $stream.Write($bytes, 0, $bytes.Length); $stream.Flush($true) } finally { $stream.Dispose() }
        Replace-Atomic $tmp $dest
    } finally { if ([IO.File]::Exists($tmp)) { [IO.File]::Delete($tmp) } }
}
function State-Set([string]$Rel, [string]$RemoteSha, [string]$LocalSha) {
    $fid = $script:Remote[$Rel].id
    $script:State[$fid] = @{path=$Rel;sha256=$RemoteSha;identity=(Local-Identity (Rel-Path $Rel))}
    $script:Manifest.tombstones.Remove($Rel) | Out-Null
    Save-State
}
function Load-State {
    $script:Migrated = $false
    $script:Manifest = @{version=2;scope=@{base_url=$script:BaseUrl;space_id=$script:SpaceId};files=(New-Dict);folders=(New-Dict);tombstones=(New-Dict);pending_deletes=(New-Dict)}
    $path = Rel-Path ([string]$Config.STATE_FILE)
    if ([IO.Directory]::Exists($path)) { Stop-Sync 'state path is a directory' }
    if ([IO.File]::Exists($path)) {
        Assert-NotLink $path
        $raw = Convert-Map ([IO.File]::ReadAllText($path, [Text.Encoding]::UTF8) | ConvertFrom-Json)
        if ($raw -isnot [Collections.IDictionary] -or (Prop $raw 'files') -isnot [Collections.IDictionary]) { Stop-Sync 'invalid state JSON' }
        if ((Prop $raw 'version') -eq 2) {
            $scope = Prop $raw 'scope'
            if ((Prop $scope 'base_url') -cne $script:BaseUrl -or (Prop $scope 'space_id') -cne $script:SpaceId) { Stop-Sync 'manifest belongs to another server/space' }
            foreach ($key in @('files','folders','tombstones','pending_deletes')) {
                if ((Prop $raw $key) -isnot [Collections.IDictionary]) { Stop-Sync "invalid manifest section $key" }
                foreach ($id in $raw[$key].Keys) {
                    $rec = $raw[$key][$id]
                    if ($rec -isnot [Collections.IDictionary]) { Stop-Sync "invalid manifest record $id" }
                    if ($key -eq 'tombstones') {
                        Assert-StatePath $id
                        if ((Prop $rec 'reason') -notin @('remote_deleted','local_deleted')) { Stop-Sync 'invalid tombstone' }
                    } else {
                        Assert-StatePath (Prop $rec 'path')
                        if ($key -ne 'folders' -and -not (Is-Sha (Prop $rec 'sha256'))) { Stop-Sync 'invalid manifest hash' }
                        if ($key -eq 'pending_deletes') { Assert-ArchivePath (Prop $rec 'archive') }
                    }
                    $script:Manifest[$key][$id] = $rec
                }
            }
        } else {
            $script:Legacy = $raw.files
            $script:Migrated = $true
            Log 'INFO legacy manifest detected: identity migration; remote delete disabled for first run'
        }
    }
    $script:State = $script:Manifest.files
}
function Api-Url([string]$Action, [hashtable]$Fields) {
    $query = @('action=' + [uri]::EscapeDataString($Action))
    foreach ($key in @($Fields.Keys | Sort-Object)) {
        $query += [uri]::EscapeDataString($key) + '=' + [uri]::EscapeDataString([string]$Fields[$key])
    }
    return $script:BaseUrl + '/api/ghost/v2/files.php?' + ($query -join '&')
}
function Api-Call {
    param(
        [string]$Method, [string]$Action,
        [hashtable]$Query = @{}, $Body = $null,
        [string]$UploadPath = '', [string]$UploadName = '', [hashtable]$Fields = @{},
        [string]$DownloadTo = '', [switch]$RetryGet
    )
    $url = Api-Url $Action $Query
    $limit = if ($Method -eq 'GET' -and $RetryGet) { $script:RetryCount } else { 1 }
    for ($attempt = 1; $attempt -le $limit; $attempt++) {
        $req = $null; $resp = $null; $cts = $null; $uploadStream = $null
        $status = 0; $errorText = ''; $transportFailed = $false
        try {
            $req = [Net.Http.HttpRequestMessage]::new([Net.Http.HttpMethod]::new($Method), $url)
            $req.Headers.Authorization = [Net.Http.Headers.AuthenticationHeaderValue]::new('Bearer', $script:Jwt)
            $req.Headers.Accept.Add([Net.Http.Headers.MediaTypeWithQualityHeaderValue]::new('application/json'))
            if ($Method -eq 'POST' -and $UploadPath) {
                $multipart = [Net.Http.MultipartFormDataContent]::new()
                foreach ($key in @($Fields.Keys)) {
                    $multipart.Add([Net.Http.StringContent]::new([string]$Fields[$key], [Text.Encoding]::UTF8), $key)
                }
                $uploadStream = [IO.File]::Open($UploadPath, [IO.FileMode]::Open, [IO.FileAccess]::Read, [IO.FileShare]::Read)
                $content = [Net.Http.StreamContent]::new($uploadStream)
                $content.Headers.ContentType = [Net.Http.Headers.MediaTypeHeaderValue]::new('application/octet-stream')
                $multipart.Add($content, 'file', $UploadName)
                $req.Content = $multipart
            } elseif ($Method -eq 'POST') {
                $json = $Body | ConvertTo-Json -Depth 10 -Compress
                $req.Content = [Net.Http.StringContent]::new($json, [Text.Encoding]::UTF8, 'application/json')
            }
            $seconds = if ($DownloadTo -or $UploadPath) { $script:TransferTimeout } else { $script:ReadTimeout }
            $cts = [Threading.CancellationTokenSource]::new([TimeSpan]::FromSeconds($seconds))
            $resp = $script:Http.SendAsync($req, [Net.Http.HttpCompletionOption]::ResponseHeadersRead, $cts.Token).GetAwaiter().GetResult()
            $status = [int]$resp.StatusCode
            if ($script:VerboseMode) { Log "HTTP $Method $url -> $status (attempt $attempt/$limit)" }
            if ($status -ge 200 -and $status -lt 300) {
                if ($DownloadTo) {
                    $inputStream = $resp.Content.ReadAsStreamAsync().GetAwaiter().GetResult()
                    $outputStream = [IO.File]::Open($DownloadTo, [IO.FileMode]::CreateNew, [IO.FileAccess]::Write, [IO.FileShare]::None)
                    try { $inputStream.CopyToAsync($outputStream, 1048576, $cts.Token).GetAwaiter().GetResult(); $outputStream.Flush($true) }
                    finally { $outputStream.Dispose(); $inputStream.Dispose() }
                    return $null
                }
                $raw = $resp.Content.ReadAsStringAsync().GetAwaiter().GetResult()
                try { $obj = $raw | ConvertFrom-Json -ErrorAction Stop }
                catch { Stop-Sync "API $Action invalid JSON at HTTP $status" }
                if ($null -eq $obj -or (Prop $obj 'ok') -cne $true) {
                    $err = Prop $obj 'error'; if (-not $err) { $err = Prop $obj 'message' }
                    Stop-Sync "API $Action returned HTTP $status without ok:true: $err"
                }
                $apiStatus = Prop $obj 'status'
                if ($apiStatus -is [string] -and $apiStatus.ToLowerInvariant() -in @('error','failed')) {
                    Stop-Sync "API $Action returned application status $apiStatus"
                }
                return $obj
            }
            $errorText = $resp.Content.ReadAsStringAsync().GetAwaiter().GetResult()
            try {
                $errObj = $errorText | ConvertFrom-Json -ErrorAction Stop
                $errField = Prop $errObj 'error'; if (-not $errField) { $errField = Prop $errObj 'message' }
                if (-not $errField) { $errField = Prop (Prop $errObj 'data') 'error' }
                if ($errField) { $errorText = [string]$errField }
            } catch {}
            if ($errorText.Length -gt 240) { $errorText = $errorText.Substring(0, 240) }
        } catch {
            # Do not retry programming, JSON, local I/O, or application validation errors.
            # Only transport/timeout errors on GET can be retried.
            $inner = $_.Exception
            while ($null -ne $inner.InnerException) { $inner = $inner.InnerException }
            if ($inner -isnot [Net.Http.HttpRequestException] -and
                $inner -isnot [OperationCanceledException] -and
                $inner -isnot [Threading.Tasks.TaskCanceledException]) { throw }
            $transportFailed = $true
            $errorText = $inner.Message
            if ($errorText.Length -gt 240) { $errorText = $errorText.Substring(0, 240) }
        } finally {
            if ($null -ne $resp) { $resp.Dispose() }
            if ($null -ne $req) { $req.Dispose() }
            if ($null -ne $cts) { $cts.Dispose() }
            if ($null -ne $uploadStream) { $uploadStream.Dispose() }
        }
        if ($status -eq 415 -and -not $transportFailed -and $Method -eq 'POST' -and
            $UploadPath -and $Action -in @('upload','file_replace')) {
            $rejection = [InvalidOperationException]::new("HTTP 415: $errorText")
            $rejection.Data['TelmiUploadRejected415'] = $true
            throw $rejection
        }
        if ($status -eq 409) { Stop-Sync "CONFLICT HTTP 409 during $Action; server wins; $errorText" }
        $retryable = ($Method -eq 'GET' -and $RetryGet -and $attempt -lt $limit -and
            ($transportFailed -or $status -eq 408 -or $status -eq 429 -or ($status -ge 500 -and $status -lt 600)))
        if ($retryable) {
            if ($DownloadTo -and [IO.File]::Exists($DownloadTo)) { [IO.File]::Delete($DownloadTo) }
            Log "WARN GET $Action HTTP $status failed; retrying ($attempt/$limit): $errorText"
            Start-Sleep -Seconds $script:RetryDelay
            continue
        }
        Stop-Sync "HTTP $Method $Action failed status=$status : $errorText"
    }
}
function Api-Data($Reply, [string]$Action) {
    $data = Prop $Reply 'data'; if ($null -eq $data) { $data = $Reply }
    if ($data -isnot [pscustomobject]) { Stop-Sync "API $Action response has no data object" }
    return $data
}
function Api-List([string]$FolderId) {
    $q = @{space_id=$script:SpaceId;limit=$script:ListLimit}
    if ($FolderId) { $q.folder_id = $FolderId }
    $d = Api-Data (Api-Call -Method 'GET' -Action 'list' -Query $q -RetryGet) 'list'
    $files = Prop $d 'files'; $folders = Prop $d 'folders'
    if ($files -isnot [array] -or $folders -isnot [array]) {
        Stop-Sync "API list folder '$FolderId' missing files[] or folders[]"
    }
    if ($files.Count -ge $script:ListLimit -or $folders.Count -ge $script:ListLimit) {
        Stop-Sync "API list folder '$FolderId' hits non-paginated limit=$($script:ListLimit) ($($files.Count) files / $($folders.Count) folders); inventory incomplete"
    }
    return @{files=$files; folders=$folders}
}
function Check-ListingParent($Parent, [string]$Expected) {
    if ($null -eq $Parent -or $Parent -ceq '' -or $Parent -ceq $Expected) { return }
    if (-not $Expected -and $Parent -ceq $script:SpaceId) { return }
    Stop-Sync "API returned mismatched folder_id/parent_id: '$Parent' (expected '$Expected')"
}
function Read-RemoteTree {
    $script:Remote = New-Dict
    $script:Folders = New-Dict
    $script:Folders.Add('', '')
    $script:Counts.folders = 0
    $seenPaths = New-Dict; $seenIds = New-Dict; $fileIds = New-Dict
    $seenPaths.Add((Canon ([string]$Config.STATE_FILE)), '[reserved state file]')
    $queue = [Collections.Generic.Queue[object]]::new()
    $queue.Enqueue(@{rel=''; id=''; depth=0})
    while ($queue.Count -gt 0) {
        $current = $queue.Dequeue()
        if ($current.depth -gt $script:MaxDepth) { Stop-Sync "remote folder depth exceeded: $($current.rel)" }
        if ($seenIds.ContainsKey('D:' + $current.id)) { Stop-Sync "repeated remote folder ID: $($current.id)" }
        $seenIds.Add('D:' + $current.id, $true)
        $listing = Api-List $current.id
        $script:Counts.folders++
        if ($script:VerboseMode) { Log "LIST $(if ($current.rel) { $current.rel } else { '/' }): $($listing.folders.Count) folders, $($listing.files.Count) files" }
        foreach ($f in $listing.folders) {
            $name = Check-Name (Prop $f 'name') 'remote folder'
            Assert-RemoteName $name
            $fid = Prop $f 'folder_id'; if ($fid -isnot [string] -or -not $fid) { Stop-Sync "API folder '$name' missing ID" }
            Check-ListingParent (Prop $f 'parent_id') $current.id
            $rel = if ($current.rel) { $current.rel + '/' + $name } else { $name }
            $canon = Canon $rel
            if ($seenPaths.ContainsKey($canon)) { Stop-Sync "remote case/Unicode/name collision: '$rel'" }
            if ($seenIds.ContainsKey('D:' + $fid) -or $script:Folders.Values -ccontains $fid) { Stop-Sync "duplicate folder ID '$fid'" }
            $seenPaths.Add($canon, $rel)
            $script:Folders.Add($rel, $fid)
            $queue.Enqueue(@{rel=$rel;id=$fid;depth=([int]$current.depth + 1)})
        }
        foreach ($f in $listing.files) {
            $name = Check-Name (Prop $f 'name') 'remote file'
            Assert-RemoteName $name
            $fid = Prop $f 'file_id'; $sha = Prop $f 'sha256'
            if ($fid -isnot [string] -or -not $fid -or $fileIds.ContainsKey($fid)) { Stop-Sync "API file '$name' has missing/duplicate file_id" }
            if (-not (Is-Sha $sha)) { Stop-Sync "API file '$name' has missing/invalid sha256" }
            Check-ListingParent (Prop $f 'folder_id') $current.id
            $rel = if ($current.rel) { $current.rel + '/' + $name } else { $name }
            $canon = Canon $rel
            if ($seenPaths.ContainsKey($canon)) { Stop-Sync "remote case/Unicode/name collision: '$rel'" }
            $seenPaths.Add($canon, $rel); $fileIds.Add($fid, $true)
            $script:Remote.Add($rel, @{id=$fid;sha=$sha.ToLowerInvariant();folder=$current.id})
        }
    }
    if ($script:Remote.Count -eq 0) { Stop-Sync 'remote inventory contains ZERO FILES; HARD STOP (no changes)' }
    Log "Remote inventory validated: $($script:Remote.Count) files, $($script:Counts.folders) folders"
}
function Read-LocalTree {
    $local = New-Dict; $pathMap = New-Dict
    $script:LocalDirs = New-Dict
    $script:LocalIds = New-Dict
    if (-not [IO.Directory]::Exists($script:Root)) { return $local }
    Assert-NotLink $script:Root
    $queue = [Collections.Generic.Queue[string]]::new(); $queue.Enqueue($script:Root)
    while ($queue.Count -gt 0) {
        $base = $queue.Dequeue()
        foreach ($entry in (Get-ChildItem -LiteralPath $base -Force -ErrorAction Stop)) {
            $name = Check-Name $entry.Name 'local file/folder'
            $full = $entry.FullName
            if (($entry.Attributes -band [IO.FileAttributes]::ReparsePoint) -ne 0) { Stop-Sync "local junction/symlink refused: $full" }
            $rel = $full.Substring($script:Root.TrimEnd('\').Length + 1).Replace('\', '/')
            $canon = Canon $rel
            if ($pathMap.ContainsKey($canon)) { Stop-Sync "local case/Unicode collision at '$rel'" }
            $pathMap.Add($canon, $rel)
            if ($entry.PSIsContainer) {
                if ($name.StartsWith('.telmi-', [StringComparison]::OrdinalIgnoreCase)) { continue }
                $script:LocalDirs[$rel] = @{identity=(Local-Identity $full)}
                $queue.Enqueue($full); continue
            }
            if (-not ($entry -is [IO.FileInfo])) { Stop-Sync "unsupported local filesystem entry: $full" }
            if (($rel -ceq [string]$Config.STATE_FILE) -or $name -ceq '.DS_Store' -or
                $name.StartsWith('.telmi-', [StringComparison]::OrdinalIgnoreCase) -or
                $name.IndexOf('.conflict-', [StringComparison]::OrdinalIgnoreCase) -ge 0) { continue }
            $local.Add($rel, (Sha $full))
            $script:LocalIds[$rel] = @{identity=(Local-Identity $full)}
        }
    }
    return $local
}
function Plan-Sync {
    $local = $script:Local
    $script:Actions = [Collections.ArrayList]::new()
    $claimed = New-Dict
    foreach ($fid in @($script:State.Keys | Sort-Object)) {
        $old = $script:State[$fid]; $lp = Find-Local $old $script:LocalIds
        if ($null -ne $lp) { $claimed[$lp] = $true }
        $rp = Remote-PathForId $fid
        if ($null -eq $rp) { $script:Counts.skipped++; continue }
        $r = $script:Remote[$rp]
        $rc = $r.sha -cne $old.sha256
        $movedServer = $rp -cne $old.path
        if ($movedServer -and $null -ne $lp -and $lp -ceq $old.path -and (Prop $old 'identity') -and $old.identity -cne $script:LocalIds[$lp].identity) { Stop-Sync 'remote file moved but local source was replaced' }
        if ($null -eq $lp) {
            if ($rc -or $movedServer -or $script:ForceRestore.ContainsKey($fid) -or $script:Mode -eq 'pull' -or $script:Migrated) {
                if ($script:Mode -ne 'push') { $null = $script:Actions.Add(@{op='download';rel=$rp;baseline=$null}) }
            } else { $null = $script:Actions.Add(@{op='delete-remote';rel=$rp;baseline=$null}) }
            continue
        }
        $lc = $local[$lp] -cne $old.sha256
        if ($movedServer -or ($lp -cne $rp -and ($rc -or $script:Mode -eq 'pull'))) {
            if ($script:Mode -eq 'push') { continue }
            if ($lp -cne $rp) { $null = $script:Actions.Add(@{op='move-local';rel=$rp;src=$lp;fid=$fid}) }
            if ($local[$lp] -cne $r.sha -and ($rc -or -not $lc)) {
                $null = $script:Actions.Add(@{op=$(if ($lc) {'conflict'} else {'download'});rel=$rp;baseline=$local[$lp]})
            }
            continue
        }
        if ($lp -cne $rp) {
            $null = $script:Actions.Add(@{op='move-remote';rel=$rp;dest=$lp;baseline=$local[$lp]}); continue
        }
        if ($local[$lp] -ceq $r.sha) {
            $null = $script:Actions.Add(@{op='adopt';rel=$rp;baseline=$r.sha})
        } elseif ($rc -and $script:Mode -ne 'push') {
            $null = $script:Actions.Add(@{op=$(if ($lc) {'conflict'} else {'download'});rel=$rp;baseline=$local[$lp]})
        } elseif ($lc -and $script:Mode -ne 'pull') {
            $null = $script:Actions.Add(@{op='replace';rel=$rp;baseline=$local[$lp]})
        }
    }
    foreach ($rel in @($script:Remote.Keys | Sort-Object)) {
        $r = $script:Remote[$rel]
        if ($script:State.ContainsKey($r.id)) { continue }
        if ($claimed.ContainsKey($rel)) { Stop-Sync "new remote ID overlaps tracked local file: $rel" }
        if ($local.ContainsKey($rel) -and $local[$rel] -ceq $r.sha) {
            $null = $script:Actions.Add(@{op='adopt';rel=$rel;baseline=$r.sha})
        } elseif ($script:Mode -ne 'push') {
            $baseline = if ($local.ContainsKey($rel)) { $local[$rel] } else { $null }
            $null = $script:Actions.Add(@{op=$(if ($null -ne $baseline) {'conflict'} else {'download'});rel=$rel;baseline=$baseline})
        }
        $claimed[$rel] = $true
    }
    foreach ($rel in @($local.Keys | Sort-Object)) {
        if ($claimed.ContainsKey($rel) -or $script:Manifest.tombstones.ContainsKey($rel) -or $script:Remote.ContainsKey($rel)) { continue }
        if ($script:Mode -ne 'pull') { $null = $script:Actions.Add(@{op='upload';rel=$rel;baseline=$local[$rel]}) }
    }
    $deletes = @($script:Actions | Where-Object { $_.op -eq 'delete-remote' })
    foreach ($act in $deletes) {
        $matches = @($script:Actions | Where-Object { $_.op -eq 'upload' -and $_.baseline -ceq $script:Remote[$act.rel].sha })
        if ($matches.Count) { Stop-Sync "ambiguous move vs delete: $($act.rel)" }
    }
    $limit = [Math]::Min($script:MaxDeletes, [Math]::Max(1, [Math]::Floor($script:Remote.Count * $script:MaxDeletePercent / 100)))
    if ($deletes.Count -gt $limit -or ($deletes.Count -gt 0 -and $deletes.Count -ge $script:Remote.Count)) { Stop-Sync 'local deletion guard exceeded or would remove final remote file' }
    Log "Plan: $($script:Actions.Count) actions; archive_deletes=$($deletes.Count)"
}
function Remote-Match([string]$Rel, $Expected) {
    $parent = Parent-Rel $Rel
    if (-not $script:Folders.ContainsKey($parent)) { Stop-Sync "unknown remote parent: $Rel" }
    $lst = Api-List ([string]$script:Folders[$parent])
    $name = Leaf-Rel $Rel
    $matches = @($lst.files | Where-Object { $n = Prop $_ 'name'; $n -is [string] -and (Canon $n) -ceq (Canon $name) })
    if ($matches.Count -gt 1) { Stop-Sync "duplicate remote filename: $Rel" }
    if ($null -eq $Expected) {
        if ($matches.Count -gt 0) { Stop-Sync "remote file appeared after inventory, server wins: $Rel" }
    } else {
        $match = if ($matches.Count -eq 1) { $matches[0] } else { $null }
        if ($null -eq $match -or (Prop $match 'name') -cne $name -or
            (Prop $match 'file_id') -cne $Expected.id -or
            (Prop $match 'sha256').ToLowerInvariant() -cne $Expected.sha) { Stop-Sync "remote changed after inventory, server wins: $Rel" }
    }
}
function Remote-EnsureFolder([string]$Rel) {
    if ($script:Folders.ContainsKey($Rel)) { return [string]$script:Folders[$Rel] }
    $parent = Parent-Rel $Rel; $name = Leaf-Rel $Rel
    $parentId = Remote-EnsureFolder $parent
    $lst = Api-List $parentId
    $matches = @($lst.folders | Where-Object { $n = Prop $_ 'name'; $n -is [string] -and (Canon $n) -ceq (Canon $name) })
    if ($matches.Count) { Stop-Sync "remote folder appeared since inventory: $Rel" }
    $body = @{space_id=$script:SpaceId;name=$name}
    if ($parentId) { $body.parent_id = $parentId }
    $d = Api-Data (Api-Call -Method 'POST' -Action 'folder_create' -Body $body) 'folder_create'
    $fid = Prop $d 'folder_id'
    if ($fid -isnot [string] -or -not $fid) { Stop-Sync "folder_create missing folder_id for $Rel" }
    $script:Folders.Add($Rel, $fid)
    Log "MKDIR-REMOTE $Rel/"
    return $fid
}
function Make-LocalFolders {
    foreach ($rel in @($script:Folders.Keys | Where-Object { $_ } | Sort-Object { ($_.ToCharArray() | Where-Object { $_ -eq '/' }).Count }, { $_ })) {
        $path = Rel-Path $rel
        if ([IO.Directory]::Exists($path)) { continue }
        if ($script:DryRun) { Log "MKDIR $rel/ (dry-run)"; continue }
        [IO.Directory]::CreateDirectory($path) | Out-Null
        if ($script:VerboseMode) { Log "MKDIR $rel/" }
    }
}
function Conflict-Copy([string]$Rel, [string]$Expected) {
    $src = Rel-Path $Rel
    if (-not [IO.File]::Exists($src)) { return }
    $sidecar = $src + '.conflict-' + (Get-Date -Format 'yyyyMMddTHHmmss') + '-' + $PID + '-' + [Guid]::NewGuid().ToString('N').Substring(0,8)
    File-CopyChecked $src $sidecar $Expected
    Log "CONFLICT-SERVER-WINS $Rel (local copy retained as $(Split-Path $sidecar -Leaf))"
    $script:Counts.conflicts++
}
function Download-Action([string]$Rel, [bool]$Conflict, $PlannedSha) {
    if ($script:DryRun) {
        Log "$(if ($Conflict) { 'CONFLICT-SERVER-WINS' } else { 'DOWNLOAD' }) $Rel (dry-run)"
        $script:Counts.downloads++; if ($Conflict) { $script:Counts.conflicts++ }; return
    }
    $record = $script:Remote[$Rel]; $dest = Rel-Path $Rel
    [IO.Directory]::CreateDirectory([IO.Path]::GetDirectoryName($dest)) | Out-Null
    $tmp = [IO.Path]::Combine([IO.Path]::GetDirectoryName($dest), '.telmi-download-' + [Guid]::NewGuid().ToString('N'))
    try {
        Api-Call -Method 'GET' -Action 'content' -Query @{file_id=$record.id} -DownloadTo $tmp -RetryGet
        $got = Sha $tmp
        if ($got -cne $record.sha) { Stop-Sync "download SHA-256 mismatch: $Rel" }
        Remote-Match $Rel $record
        $nowSha = $null
        if ([IO.File]::Exists($dest)) { $nowSha = Sha $dest }
        if ($null -ne $nowSha -and $nowSha -cne $got -and ($Conflict -or $nowSha -cne $PlannedSha)) {
            Conflict-Copy $Rel $nowSha
        }
        if ([IO.File]::Exists($dest)) { Assert-NotLink $dest }
        Replace-Atomic $tmp $dest
        State-Set $Rel $got $got
        Log "DOWNLOAD $Rel"; $script:Counts.downloads++
    } finally { if ([IO.File]::Exists($tmp)) { [IO.File]::Delete($tmp) } }
}
function Upload-Action([string]$Rel, [string]$PlannedSha, [bool]$NewFile) {
    $label = if ($NewFile) { 'UPLOAD' } else { 'REPLACE' }
    if ($script:DryRun) { Log "$label $Rel (dry-run)"; $script:Counts.uploads++; return }
    $src = Rel-Path $Rel
    if (-not [IO.File]::Exists($src) -or (Sha $src) -cne $PlannedSha) { Stop-Sync "local file changed since plan: $Rel" }
    $parent = Parent-Rel $Rel; $name = Leaf-Rel $Rel
    $folderId = Remote-EnsureFolder $parent
    $record = $null; if (-not $NewFile) { $record = $script:Remote[$Rel] }
    Remote-Match $Rel $record
    $snap = [IO.Path]::Combine($script:WorkDir, 'upload-' + [Guid]::NewGuid().ToString('N'))
    try {
        File-CopyChecked $src $snap $PlannedSha
        $fields = @{}
        if ($NewFile) {
            $fields.space_id = $script:SpaceId
            if ($folderId) { $fields.folder_id = $folderId }
        } else {
            $fields.file_id = $record.id
            $fields.expected_sha256 = $record.sha
        }
        $action = if ($NewFile) { 'upload' } else { 'file_replace' }
        $d = Api-Data (Api-Call -Method 'POST' -Action $action -UploadPath $snap -UploadName $name -Fields $fields) $action
        $reported = Prop $d 'sha256'
        if ($null -ne $reported -and (-not (Is-Sha $reported) -or $reported.ToLowerInvariant() -cne $PlannedSha)) {
            Stop-Sync "API $action reported wrong SHA-256 for $Rel"
        }
        $returnedId = Prop $d 'file_id'
        if (-not $NewFile -and $null -ne $returnedId -and $returnedId -cne $record.id) { Stop-Sync "file_replace unexpectedly changed file_id for $Rel" }
        $lst = Api-List $folderId
        $matches = @($lst.files | Where-Object { $n = Prop $_ 'name'; $n -is [string] -and (Canon $n) -ceq (Canon $name) })
        if ($matches.Count -ne 1) { Stop-Sync "post-$action verification found $($matches.Count) matching files for $Rel" }
        $f = $matches[0]; $newSha = Prop $f 'sha256'; $newId = Prop $f 'file_id'
        if ((Prop $f 'name') -cne $name -or $newId -isnot [string] -or -not $newId) { Stop-Sync "post-$action returned unexpected name or missing file_id for $Rel" }
        if ((-not (Is-Sha $newSha)) -or $newSha.ToLowerInvariant() -cne $PlannedSha) { Stop-Sync "post-$action checksum mismatch for $Rel" }
        if (-not $NewFile -and $newId -cne $record.id) { Stop-Sync "post-file_replace ID changed for $Rel" }
        $script:Remote[$Rel] = @{id=$newId;sha=$PlannedSha;folder=$folderId}
        State-Set $Rel $PlannedSha $PlannedSha
        Log "$label $Rel"; $script:Counts.uploads++
    } catch {
        if ($_.Exception.Data['TelmiUploadRejected415'] -eq $true) {
            Log "SKIP $Rel ($($_.Exception.Message))"
            $script:Counts.skipped++
            return
        }
        throw
    } finally { if ([IO.File]::Exists($snap)) { [IO.File]::Delete($snap) } }
}
function Run-Sync {
    $script:ForceRestore = New-Dict
    Load-State
    Read-RemoteTree
    $script:Local = Read-LocalTree
    if ($script:Migrated) {
        foreach ($rel in $script:Legacy.Keys) {
            Assert-StatePath $rel
            $entry = $script:Legacy[$rel]
            if (-not (Is-Sha (Prop $entry 'remote_sha256')) -or -not (Is-Sha (Prop $entry 'local_sha256'))) { Stop-Sync 'invalid legacy hashes' }
            if ($script:Remote.ContainsKey($rel)) {
                if (-not $script:Local.ContainsKey($rel)) { Stop-Sync "legacy migration blocked: previously synced local file missing: $rel" }
                $r = $script:Remote[$rel]
                $script:State[$r.id] = @{path=$rel;sha256=$entry.remote_sha256;identity=$script:LocalIds[$rel].identity}
            } else { $script:Manifest.tombstones[$rel] = @{reason='remote_deleted';sha256=$entry.remote_sha256} }
        }
    }
    Assert-CrossPaths
    Recover-Deletes
    if (Run-FolderMoves) { return }
    Read-RemoteTree
    $script:Local = Read-LocalTree
    Make-LocalFolders
    Apply-RemoteDeletes
    Plan-Sync
    foreach ($act in @($script:Actions | Where-Object { $_.op -ne 'delete-remote' }) + @($script:Actions | Where-Object { $_.op -eq 'delete-remote' })) {
        switch ($act.op) {
            'adopt' {
                if (-not $script:DryRun) {
                    if ((Sha (Rel-Path $act.rel)) -cne $act.baseline) { Stop-Sync 'local changed before adopt' }
                    State-Set $act.rel $act.baseline $act.baseline
                }
            }
            'download' { Download-Action $act.rel $false $act.baseline }
            'conflict' { Download-Action $act.rel $true $act.baseline }
            'upload' { Upload-Action $act.rel $act.baseline $true }
            'replace' { Upload-Action $act.rel $act.baseline $false }
            'move-local' {
                Move-Local $act.src $act.rel $false
                if (-not $script:DryRun) {
                    $script:State[$act.fid].path = $act.rel
                    $script:State[$act.fid].identity = Local-Identity (Rel-Path $act.rel)
                    Save-State
                }
            }
            'move-remote' {
                if ($script:DryRun) { Log "MOVE-REMOTE-FILE $($act.rel) -> $($act.dest) (dry-run)"; continue }
                $r = $script:Remote[$act.rel]
                if (-not [IO.File]::Exists((Rel-Path $act.dest)) -or (Sha (Rel-Path $act.dest)) -cne $act.baseline) { Stop-Sync 'local move destination changed before file_update' }
                Remote-Match $act.rel $r
                $folder = Remote-EnsureFolder (Parent-Rel $act.dest)
                Remote-Match $act.dest $null
                $null = Api-Call -Method POST -Action file_update -Body @{file_id=$r.id;name=(Leaf-Rel $act.dest);folder_id=$(if ($folder) {$folder} else {$null})}
                $script:Remote.Remove($act.rel) | Out-Null
                $script:Remote[$act.dest] = @{id=$r.id;sha=$r.sha;folder=$folder}
                Remote-Match $act.dest $script:Remote[$act.dest]
                State-Set $act.dest $r.sha $act.baseline
                Log "MOVE-REMOTE-FILE $($act.rel) -> $($act.dest) (file_id preserved)"
                $script:Counts.moves++
            }
            'delete-remote' { Delete-Remote $act.rel }
            default { Stop-Sync "unknown action $($act.op)" }
        }
    }
    if (-not $script:DryRun) { Refresh-FolderState; Save-State }
    Log ('telmi-files-sync done: ' + (($script:Counts.Keys | Sort-Object | ForEach-Object { "$_=$($script:Counts[$_])" }) -join ' ') + ' errors=0')
}

# v6 identity/move/delete helpers. File identities survive ordinary NTFS renames.
function Convert-Map($Value) {
    if ($Value -is [pscustomobject]) {
        $result = New-Dict
        foreach ($p in $Value.PSObject.Properties) { $result[$p.Name] = Convert-Map $p.Value }
        return ,$result
    }
    return ,$Value
}
function Assert-StatePath($Rel) {
    if ($Rel -isnot [string] -or -not $Rel) { Stop-Sync 'invalid manifest path' }
    $null = Rel-Path $Rel
    foreach ($part in $Rel.Split('/')) {
        if ($part.StartsWith('.telmi-', [StringComparison]::OrdinalIgnoreCase) -or $part -ceq [string]$Config.STATE_FILE) { Stop-Sync 'reserved manifest path' }
    }
}
function Assert-ArchivePath($Rel) {
    if ($Rel -isnot [string] -or -not $Rel.StartsWith('.telmi-deleted/', [StringComparison]::Ordinal)) { Stop-Sync 'invalid archive path' }
    $null = Rel-Path $Rel
}
function Assert-RemoteName([string]$Name) {
    if ($Name.StartsWith('.telmi-', [StringComparison]::OrdinalIgnoreCase) -or $Name -ceq '.DS_Store' -or $Name.IndexOf('.conflict-', [StringComparison]::OrdinalIgnoreCase) -ge 0) { Stop-Sync "remote name would be excluded: $Name" }
}
function Local-Identity([string]$Path) { return [TelmiFileIdentity]::Get($Path) }
function Find-Local($Record, $Paths) {
    $identity = Prop $Record 'identity'
    $hits = @($Paths.Keys | Where-Object { $identity -and (Prop $Paths[$_] 'identity') -ceq $identity })
    if ($hits.Count -gt 1) { Stop-Sync 'filesystem identity at multiple paths; hard links unsupported' }
    if ($hits.Count -eq 1) { return [string]$hits[0] }
    if ($Paths.ContainsKey($Record.path)) { return [string]$Record.path }
    return $null
}
function Remote-PathForId([string]$Id) {
    foreach ($p in $script:Remote.Keys) { if ($script:Remote[$p].id -ceq $Id) { return [string]$p } }
    return $null
}
function Assert-CrossPaths {
    $names = New-Dict
    foreach ($p in @($script:Remote.Keys) + @($script:Folders.Keys)) { $names[(Canon $p)] = $p }
    foreach ($p in @($script:Local.Keys) + @($script:LocalDirs.Keys)) {
        if ($names.ContainsKey((Canon $p)) -and $names[(Canon $p)] -cne $p) { Stop-Sync "case/Unicode mismatch: $p" }
        if (($script:Remote.ContainsKey($p) -and $script:LocalDirs.ContainsKey($p)) -or ($script:Folders.ContainsKey($p) -and $script:Local.ContainsKey($p))) { Stop-Sync "file/folder collision: $p" }
    }
}
function Move-Local([string]$Src, [string]$Dest, [bool]$Directory) {
    $from = Rel-Path $Src; $to = Rel-Path $Dest
    if ([IO.File]::Exists($to) -or [IO.Directory]::Exists($to)) { Stop-Sync "move destination occupied: $Dest" }
    if ($script:DryRun) { Log "MOVE-LOCAL $Src -> $Dest (dry-run)"; return }
    [IO.Directory]::CreateDirectory([IO.Path]::GetDirectoryName($to)) | Out-Null
    if ($Directory) { [IO.Directory]::Move($from, $to) } else { [IO.File]::Move($from, $to) }
    $script:Counts.moves++
    Log "MOVE-LOCAL $Src -> $Dest"
}
function Rewrite-StatePaths([string]$From, [string]$To) {
    foreach ($section in @('files','folders')) {
        foreach ($rec in $script:Manifest[$section].Values) {
            if ($rec.path -ceq $From) { $rec.path = $To }
            elseif ($rec.path.StartsWith($From + '/', [StringComparison]::Ordinal)) { $rec.path = $To + $rec.path.Substring($From.Length) }
        }
    }
    foreach ($p in @($script:Manifest.tombstones.Keys)) {
        if ($p -ceq $From -or $p.StartsWith($From + '/', [StringComparison]::Ordinal)) {
            $dest = $To + $p.Substring($From.Length)
            if ($script:Manifest.tombstones.ContainsKey($dest)) { Stop-Sync 'tombstone collision after folder move' }
            $script:Manifest.tombstones[$dest] = $script:Manifest.tombstones[$p]
            $script:Manifest.tombstones.Remove($p) | Out-Null
        }
    }
}
function Refresh-FolderState {
    $null = Read-LocalTree
    $seen = New-Dict
    foreach ($p in $script:Folders.Keys) {
        if (-not $p) { continue }
        $id = [string]$script:Folders[$p]; $seen[$id] = $true
        $identity = if ($script:LocalDirs.ContainsKey($p)) { $script:LocalDirs[$p].identity } else { $null }
        $script:Manifest.folders[$id] = @{path=$p;identity=$identity}
    }
    foreach ($id in @($script:Manifest.folders.Keys)) { if (-not $seen.ContainsKey($id)) { $script:Manifest.folders.Remove($id) | Out-Null } }
}
function Run-FolderMoves {
    $rounds = $script:Manifest.folders.Count + 3
    for ($round = 0; $round -lt $rounds; $round++) {
        $changed = $false
        foreach ($fid in @($script:Manifest.folders.Keys | Sort-Object { $script:Manifest.folders[$_].path.Length })) {
            $old = $script:Manifest.folders[$fid]; $rp = $null
            foreach ($p in $script:Folders.Keys) { if ($script:Folders[$p] -ceq $fid) { $rp = $p; break } }
            if ($null -eq $rp) { continue }
            $lp = Find-Local $old $script:LocalDirs
            if ($rp -cne $old.path) {
                if ($script:Mode -eq 'push') { continue }
                if ($null -ne $lp -and $lp -ceq $old.path -and (Prop $old 'identity') -and $old.identity -cne $script:LocalDirs[$lp].identity) { Stop-Sync 'remote folder moved but local source was replaced' }
                if ($null -eq $lp) {
                    foreach ($p in $script:Remote.Keys) { if ($p.StartsWith($rp + '/', [StringComparison]::Ordinal)) { $script:ForceRestore[$script:Remote[$p].id] = $true } }
                }
                if ($null -ne $lp -and $lp -cne $rp) { Move-Local $lp $rp $true }
                if ($script:DryRun) { Log 'DRY-RUN: folder move listed; rerun after applying it for remaining file actions'; return $true }
                Rewrite-StatePaths $old.path $rp
                $changed = $true
            } elseif ($null -ne $lp -and $lp -cne $rp) {
                if ($script:Mode -eq 'pull') { Move-Local $lp $rp $true }
                else {
                    if ($lp.StartsWith($rp + '/', [StringComparison]::Ordinal)) { Stop-Sync 'folder cannot move into itself' }
                    if ($script:DryRun) { Log "MOVE-REMOTE-FOLDER $rp -> $lp (dry-run)"; return $true }
                    $parent = Remote-EnsureFolder (Parent-Rel $lp)
                    Read-RemoteTree
                    if (-not $script:Folders.ContainsKey($rp) -or $script:Folders[$rp] -cne $fid -or $script:Folders.ContainsKey($lp)) { Stop-Sync 'remote folder changed before move' }
                    $null = Api-Call -Method POST -Action folder_update -Body @{folder_id=$fid;name=(Leaf-Rel $lp);parent_id=$(if ($parent) {$parent} else {$null})}
                    Read-RemoteTree
                    if (-not $script:Folders.ContainsKey($lp) -or $script:Folders[$lp] -cne $fid) { Stop-Sync 'folder move verification failed' }
                    Rewrite-StatePaths $rp $lp
                    $script:Counts.moves++
                    Log "MOVE-REMOTE-FOLDER $rp -> $lp (folder_id preserved)"
                }
                if ($script:DryRun) { return $true }
                $changed = $true
            }
            if ($changed) {
                Save-State
                Read-RemoteTree
                $script:Local = Read-LocalTree
                break
            }
        }
        if (-not $changed) { return $false }
    }
    Stop-Sync 'folder move detection did not converge'
}
function Apply-RemoteDeletes {
    if ($script:Mode -eq 'push') { return }
    $candidates = New-Dict; $missing = @()
    foreach ($fid in @($script:State.Keys)) {
        if ($null -eq (Remote-PathForId $fid)) {
            $missing += $fid
            $lp = Find-Local $script:State[$fid] $script:LocalIds
            if ($null -ne $lp) { $candidates[$lp] = $true }
        }
    }
    $oldTombstones = @()
    foreach ($p in @($script:Manifest.tombstones.Keys)) {
        $rec = $script:Manifest.tombstones[$p]
        if ($rec.reason -ceq 'remote_deleted' -and -not $script:Remote.ContainsKey($p) -and $null -eq (Remote-PathForId ([string](Prop $rec 'file_id')))) {
            $oldTombstones += $p
            if ($script:Local.ContainsKey($p)) { $candidates[$p] = $true }
        }
    }
    foreach ($p in $candidates.Keys) {
        $path = Rel-Path $p
        if (-not [IO.File]::Exists($path)) { Stop-Sync "remote deletion target changed: $p" }
        if (-not $script:DryRun) { [IO.File]::Delete($path) }
        Log "DELETE-LOCAL $p (server deleted; no archive)$(if ($script:DryRun) {' (dry-run)'})"
        $script:Counts.local_deletes++
        $script:Local.Remove($p) | Out-Null
        $script:LocalIds.Remove($p) | Out-Null
    }
    foreach ($fid in $missing) { $script:State.Remove($fid) | Out-Null }
    foreach ($p in $oldTombstones) { $script:Manifest.tombstones.Remove($p) | Out-Null }
    Save-State
}
function Finish-Delete([string]$Fid) {
    $pending = $script:Manifest.pending_deletes[$Fid]
    $script:Manifest.tombstones[$pending.path] = @{reason='local_deleted';file_id=$Fid;sha256=$pending.sha256;archive=$pending.archive}
    $script:Manifest.pending_deletes.Remove($Fid) | Out-Null
    $script:State.Remove($Fid) | Out-Null
    Save-State
}
function Recover-Deletes {
    foreach ($fid in @($script:Manifest.pending_deletes.Keys)) {
        $pending = $script:Manifest.pending_deletes[$fid]
        $path = Rel-Path $pending.archive
        if (-not [IO.File]::Exists($path) -or (Sha $path) -cne $pending.sha256) { Stop-Sync 'pending deletion archive missing/corrupt' }
        if ($null -ne (Remote-PathForId $fid)) { Stop-Sync 'pending delete still exists remotely; archive safe; manual resolution required' }
        if (-not $script:DryRun) { Finish-Delete $fid }
        Log "RECOVER-COMMITTED-DELETE $($pending.path)"
    }
}
function Delete-Remote([string]$Rel) {
    if ($script:DryRun) { Log "ARCHIVE+DELETE-REMOTE $Rel (dry-run)"; $script:Counts.deletes++; return }
    $remote = $script:Remote[$Rel]
    Read-RemoteTree
    if ($script:Remote.Count -le 1) { Stop-Sync 'delete would leave remote space empty' }
    if ([IO.File]::Exists((Rel-Path $Rel)) -or [IO.Directory]::Exists((Rel-Path $Rel))) { Stop-Sync 'local file reappeared before delete' }
    Remote-Match $Rel $remote
    $archive = '.telmi-deleted/' + (Get-Date -Format yyyy-MM-dd) + '/' + [Guid]::NewGuid().ToString('N') + '/' + $Rel
    $dest = Rel-Path $archive
    [IO.Directory]::CreateDirectory([IO.Path]::GetDirectoryName($dest)) | Out-Null
    $archiveRoot = [IO.DirectoryInfo]::new((Rel-Path '.telmi-deleted'))
    $archiveRoot.Attributes = $archiveRoot.Attributes -bor [IO.FileAttributes]::Hidden
    Api-Call -Method GET -Action content -Query @{file_id=$remote.id} -DownloadTo $dest -RetryGet
    $flush = [IO.File]::Open($dest, 'Open', 'ReadWrite', 'None')
    try { $flush.Flush($true) } finally { $flush.Dispose() }
    if ((Sha $dest) -cne $remote.sha) { Stop-Sync 'archive checksum mismatch; delete refused' }
    Remote-Match $Rel $remote
    if ([IO.File]::Exists((Rel-Path $Rel)) -or [IO.Directory]::Exists((Rel-Path $Rel))) { Stop-Sync 'local file reappeared during archive' }
    $script:Manifest.pending_deletes[$remote.id] = @{path=$Rel;sha256=$remote.sha;archive=$archive}
    Save-State
    $null = Api-Call -Method POST -Action file_delete -Body @{file_id=$remote.id}
    $listing = Api-List ([string]$script:Folders[(Parent-Rel $Rel)])
    if (@($listing.files | Where-Object { (Prop $_ 'file_id') -ceq $remote.id }).Count) { Stop-Sync 'delete verification failed' }
    Finish-Delete $remote.id
    $script:Counts.deletes++
    Log "DELETE-REMOTE $Rel (archived $archive)"
}

# --- MAIN --------------------------------------------------------------------
$script:LogPath = $null
$script:Http = $null
$script:LockStream = $null
$script:WorkDir = $null
$exitCode = 0
try {
    $script:DryRun = $false; $script:Mode = 'both'; $script:VerboseMode = $false
    $pull = $false; $push = $false
    foreach ($arg in $args) {
        switch ([string]$arg) {
            '--dry-run' { $script:DryRun = $true }
            '--pull' { $pull = $true }
            '--push' { $push = $true }
            '--verbose' { $script:VerboseMode = $true }
            '-v' { $script:VerboseMode = $true }
            '--help' { Write-Host 'Usage: .\telmi-files-sync-windows.ps1 [--dry-run] [--pull] [--push] [--verbose]'; exit 0 }
            '-h' { Write-Host 'Usage: .\telmi-files-sync-windows.ps1 [--dry-run] [--pull] [--push] [--verbose]'; exit 0 }
            default { Stop-Sync "Unknown option: $arg" }
        }
    }
    if ($pull -and $push) { Stop-Sync '--pull and --push are mutually exclusive' }
    if ($pull) { $script:Mode = 'pull' }; if ($push) { $script:Mode = 'push' }
    if ([string]::IsNullOrWhiteSpace($env:LOCALAPPDATA) -or [string]::IsNullOrWhiteSpace($env:USERPROFILE)) { Stop-Sync 'Windows user profile unavailable' }
    # Establish the persistent log before validating credentials/settings, so
    # even configuration errors produce a logged hard stop.
    [IO.Directory]::CreateDirectory([string]$Config.LOG_DIR) | Out-Null
    $script:LogPath = [IO.Path]::Combine([string]$Config.LOG_DIR, 'telmi-files-sync.log')
    $script:Jwt = [string]$Config.TELMI_GHOST_JWT
    $script:SpaceId = [string]$Config.TELMI_SPACE_ID
    $script:BaseUrl = ([string]$Config.TELMI_BASE_URL).TrimEnd('/')
    if ($script:Jwt -eq 'PASTE_YOUR_GHOST_JWT_HERE' -or [string]::IsNullOrWhiteSpace($script:Jwt)) { Stop-Sync 'TELMI_GHOST_JWT not configured' }
    if ($script:SpaceId -eq 'PASTE_YOUR_SPACE_ID_HERE' -or $script:SpaceId -cnotmatch '^[A-Za-z0-9_-]{1,128}$') { Stop-Sync 'TELMI_SPACE_ID missing/invalid' }
    $uri = $null
    if (-not [uri]::TryCreate($script:BaseUrl, [UriKind]::Absolute, [ref]$uri) -or $uri.Scheme -cne 'https' -or
        $uri.UserInfo -or $uri.Query -or $uri.Fragment -or $uri.AbsolutePath -ne '/') { Stop-Sync 'TELMI_BASE_URL must be an HTTPS origin (no path/query/credentials)' }
    $null = Check-Name ([string]$Config.STATE_FILE) 'STATE_FILE'
    $script:Root = [IO.Path]::GetFullPath([Environment]::ExpandEnvironmentVariables([string]$Config.SYNC_DIR)).TrimEnd('\')
    if ($script:Root.Length -lt 4) { Stop-Sync 'unsafe sync root directory' }
    if ([IO.File]::Exists($script:Root)) { Stop-Sync 'SYNC_DIR is a file' }
    if ([IO.Directory]::Exists($script:Root)) { Assert-NotLink $script:Root }
    $script:ListLimit = Parse-Int 'LIST_PAGE_SIZE' 1 500
    $script:MaxDepth = Parse-Int 'MAX_FOLDER_DEPTH' 1 128
    $script:RetryCount = Parse-Int 'HTTP_RETRIES' 1 20
    $script:RetryDelay = Parse-Int 'HTTP_RETRY_SLEEP' 0 600
    $script:ReadTimeout = Parse-Int 'HTTP_READ_TIMEOUT' 1 86400
    $script:TransferTimeout = Parse-Int 'HTTP_TRANSFER_TIMEOUT' 1 86400
    $script:MaxDeletes = Parse-Int 'MAX_LOCAL_DELETES_PER_RUN' 1 100000
    $script:MaxDeletePercent = Parse-Int 'MAX_LOCAL_DELETE_PERCENT' 1 100
    $script:Counts = @{folders=0;downloads=0;uploads=0;conflicts=0;skipped=0;moves=0;deletes=0;local_deletes=0}

if (-not ('TelmiFileIdentity' -as [type])) {
    Add-Type -TypeDefinition @'
using System;
using System.ComponentModel;
using System.Runtime.InteropServices;
using Microsoft.Win32.SafeHandles;
public static class TelmiFileIdentity {
    [StructLayout(LayoutKind.Sequential)] struct Info {
        public uint Attributes; public System.Runtime.InteropServices.ComTypes.FILETIME Creation, Access, Write;
        public uint Volume, SizeHigh, SizeLow, Links, IndexHigh, IndexLow;
    }
    [DllImport("kernel32.dll", CharSet=CharSet.Unicode, SetLastError=true)]
    static extern SafeFileHandle CreateFile(string name, uint access, uint share, IntPtr security, uint mode, uint flags, IntPtr template);
    [DllImport("kernel32.dll", SetLastError=true)]
    static extern bool GetFileInformationByHandle(SafeFileHandle handle, out Info info);
    public static string Get(string path) {
        using (SafeFileHandle h = CreateFile(path, 0, 7, IntPtr.Zero, 3, 0x02000000, IntPtr.Zero)) {
            if (h.IsInvalid) throw new Win32Exception(Marshal.GetLastWin32Error());
            Info info;
            if (!GetFileInformationByHandle(h, out info)) throw new Win32Exception(Marshal.GetLastWin32Error());
            return info.Volume.ToString("X8") + ":" + info.IndexHigh.ToString("X8") + info.IndexLow.ToString("X8");
        }
    }
}
'@
}

    Log "telmi-files-sync start mode=$($script:Mode) dry_run=$([int]$script:DryRun) space=$($script:SpaceId)"
    # Exclusive OS file lock: held open with FileShare.None; automatically
    # released on process exit/crash, no unsafe stale-lock deletion or PID reuse.
    [IO.Directory]::CreateDirectory([string]$Config.LOCK_DIR) | Out-Null
    $lockInput = $script:Root.ToUpperInvariant()
    $lockBytes = [Text.Encoding]::UTF8.GetBytes($lockInput)
    $hashAlgo = [Security.Cryptography.SHA256]::Create()
    try { $lockName = ([BitConverter]::ToString($hashAlgo.ComputeHash($lockBytes))).Replace('-', '').Substring(0,24) }
    finally { $hashAlgo.Dispose() }
    $lockPath = [IO.Path]::Combine([string]$Config.LOCK_DIR, "space-$lockName.lock")
    try { $script:LockStream = [IO.File]::Open($lockPath, [IO.FileMode]::OpenOrCreate, [IO.FileAccess]::ReadWrite, [IO.FileShare]::None) }
    catch [IO.IOException] { Stop-Sync "another sync is running or lock unavailable ($lockPath)" }
    $script:WorkDir = [IO.Path]::Combine([IO.Path]::GetTempPath(), 'telmi-files-sync-' + [Guid]::NewGuid().ToString('N'))
    [IO.Directory]::CreateDirectory($script:WorkDir) | Out-Null
    Add-Type -AssemblyName System.Net.Http
    [Net.ServicePointManager]::SecurityProtocol = [Net.ServicePointManager]::SecurityProtocol -bor [Net.SecurityProtocolType]::Tls12
    $handler = [Net.Http.HttpClientHandler]::new()
    $handler.AllowAutoRedirect = $false  # don't disclose bearer JWT via redirects
    $script:Http = [Net.Http.HttpClient]::new($handler)
    $script:Http.Timeout = [Threading.Timeout]::InfiniteTimeSpan  # per-request cancellation tokens
    Run-Sync
} catch {
    $exitCode = 1
    $message = $_.Exception.Message
    try { Log "ERROR HARD STOP: $message" }
    catch { [Console]::Error.WriteLine("ERROR HARD STOP: $message (logging also failed)") }
} finally {
    if ($null -ne $script:Http) { $script:Http.Dispose() }
    if ($null -ne $script:LockStream) { $script:LockStream.Dispose() }
    if ($script:WorkDir -and [IO.Directory]::Exists($script:WorkDir)) {
        try { [IO.Directory]::Delete($script:WorkDir, $true) }
        catch { [Console]::Error.WriteLine("WARN could not remove temporary work folder: $($script:WorkDir)") }
    }
}
exit $exitCode
